# Runtime → Change runtime type → T4 GPU (иначе модель поедет на CPU и будет
# отвечать минутами).
#
# Ячейка:
#   1. ставит Ollama и тянет модель,
#   2. поднимает FastAPI с эндпоинтом POST /generate,
#   3. открывает Cloudflare-туннель и печатает публичный URL.
#
# Дальше скопируй напечатанный URL и токен в локальный .env:
#   AGENT_COLAB_URL=https://<...>.trycloudflare.com
#   AGENT_COLAB_TOKEN=<напечатанный токен>
#
# ВАЖНО: адрес туннеля меняется при КАЖДОМ перезапуске ячейки — .env надо
# обновлять после каждого рестарта. Ячейка живёт, пока открыта вкладка Colab.
#
# Контракт эндпоинта (его ждёт code_agent/planners/colab.py):
#   запрос : {"text": "<промпт>", "token": "<AGENT_COLAB_TOKEN>"}
#   ответ  : {"response": "<сырой текст модели>"}
# =============================================================================

# --- 0. Настройки ------------------------------------------------------------
MODEL = "gpt-oss:20b"   # можно заменить на llama3.1:8b / qwen2.5-coder:7b и т.п.
PORT = 8000

# --- 2. Запуск Ollama и загрузка модели --------------------------------------
import subprocess, time, os, secrets, threading, re, requests

subprocess.Popen(["ollama", "serve"], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)

for _ in range(60):
    try:
        requests.get("http://127.0.0.1:11434/api/tags", timeout=2)
        break
    except Exception:
        time.sleep(1)
else:
    raise RuntimeError("Ollama не поднялся за 60 секунд")

print(f"⬇️  Тяну модель {MODEL} (первый раз это долго)...")
subprocess.run(["ollama", "pull", MODEL], check=True)
print("✅ Модель готова")

# --- 3. Токен ----------------------------------------------------------------
# Туннель публичный: без токена твоей GPU может пользоваться кто угодно,
# кто угадает адрес. Токен генерируется на каждый запуск ячейки.
TOKEN = os.environ.get("AGENT_COLAB_TOKEN") or secrets.token_urlsafe(24)

# --- 4. FastAPI --------------------------------------------------------------
import nest_asyncio, uvicorn
from fastapi import FastAPI
from fastapi.responses import JSONResponse
from pydantic import BaseModel

nest_asyncio.apply()
app = FastAPI(title="colabagent-free server")


class GenerateRequest(BaseModel):
    text: str = ""
    token: str = ""


@app.get("/health")
def health():
    return {"status": "ok", "model": MODEL}


@app.post("/generate")
def generate(req: GenerateRequest):
    # Сравнение в постоянное время — чтобы токен нельзя было подобрать
    # побайтово по времени ответа.
    if not secrets.compare_digest(req.token or "", TOKEN):
        return JSONResponse({"error": "unauthorized"}, status_code=401)

    if not req.text.strip():
        return JSONResponse({"error": "empty prompt"}, status_code=400)

    try:
        r = requests.post(
            "http://127.0.0.1:11434/api/generate",
            json={
                "model": MODEL,
                "prompt": req.text,
                "stream": False,
                "options": {"temperature": 0.3, "num_predict": 512},
            },
            timeout=300,
        )
        r.raise_for_status()
        # Планировщик ждёт именно поле "response" с сырым текстом модели —
        # разбор JSON и снятие ```-обёртки происходит на стороне агента.
        return {"response": r.json().get("response", "")}
    except Exception as e:
        return JSONResponse({"error": str(e)[:300]}, status_code=502)


threading.Thread(
    target=lambda: uvicorn.run(app, host="0.0.0.0", port=PORT, log_level="warning"),
    daemon=True,
).start()
time.sleep(3)

# --- 5. Cloudflare-туннель ---------------------------------------------------
tunnel = subprocess.Popen(
    ["cloudflared", "tunnel", "--url", f"http://127.0.0.1:{PORT}", "--no-autoupdate"],
    stdout=subprocess.PIPE,
    stderr=subprocess.STDOUT,
    text=True,
    bufsize=1,
)

public_url = None
for line in tunnel.stdout:
    m = re.search(r"https://[-\w]+\.trycloudflare\.com", line)
    if m:
        public_url = m.group(0)
        break

if not public_url:
    raise RuntimeError("Не удалось получить URL туннеля — посмотри вывод cloudflared выше")

print("\n" + "=" * 68)
print("✅ Сервер поднят. Впиши это в локальный .env:\n")
print(f"AGENT_COLAB_URL={public_url}")
print(f"AGENT_COLAB_TOKEN={TOKEN}")
print("=" * 68)
print("\nПроверка со своей машины:")
print(f'  curl {public_url}/health')
print(f'''  curl -X POST {public_url}/generate \\
       -H "Content-Type: application/json" \\
       -d '{{"text":"скажи привет","token":"{TOKEN}"}}' ''')
print("\n⚠️  Не закрывай вкладку: туннель живёт только пока ячейка выполняется.")

# Держим ячейку живой.
while True:
    time.sleep(60)
